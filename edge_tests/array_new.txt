routine checkArray (): array[3] real is 
    var new: array[3] real;
    new[2] := 100;
    new[1] := 150;
    new[3] := 130;
    return new;
end


routine main(): real is
    var asd: array[3] real;
    asd := checkArray();
    var abd: array[10] real;
    abd := checkArray();
    return asd[3] + abd[1];
end
